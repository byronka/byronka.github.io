54      /**
55       * FINDING 11: the script close-tag match is case sensitive and whitespace intolerant,
56       * so the rest of the document is silently swallowed.
57       * <p>
58       *     Defect: {@code src/main/java/com/renomad/minum/htmlparsing/HtmlParser.java:275} and
59       *     {@code :282} - {@code scriptElement} is the literal lowercase character list
60       *     {@code </script>} and {@code determineScriptState} looks for exactly that sequence in
61       *     the ring buffer.  Opening tags, by contrast, are matched case-insensitively, so
62       *     {@code <SCRIPT>} does put the parser into script mode.  The result is that
63       *     {@code </SCRIPT>}, or {@code </script >} (a space before the angle bracket, which HTML
64       *     permits), never ends script mode.  The parser then consumes the entire remainder of
65       *     the document as script text and returns an empty node list - with no exception and no
66       *     diagnostic.  Anything that relies on this parser to inspect a document (the
67       *     framework's own functional-testing helpers, or any application scanning HTML) is
68       *     silently handed nothing.
69       * </p>
70       * <p>
71       *     Correct behavior: the closing script tag must be recognised case-insensitively and
72       *     must tolerate whitespace before the closing angle bracket, exactly as the opening tag
73       *     is.  In each of the three documents below the trailing {@code <p>important</p>} must
74       *     still be parsed.
75       * </p>
76       * // BK: initial thought - the html parser is intended for use on trusted documents, thus invalid test.
77       */
78      @Ignore("the html parser is intended for use on trusted documents, thus invalid test")
79      public void test_Finding_ScriptCloseTagIsCaseInsensitive() {
80          var htmlParser = new HtmlParser();
81  
82          // control: the all-lowercase, whitespace-free form works today
83          List<HtmlParseNode> control = htmlParser.parse("<script>var x = 1;</script><p>important</p>");
84          assertEquals(control.size(), 2);
85  
86          assertParagraphIsFound(htmlParser, "<SCRIPT>var x = 1;</SCRIPT><p>important</p>",
87                  "an uppercase <SCRIPT> ... </SCRIPT> block");
88          assertParagraphIsFound(htmlParser, "<script>var x = 1;</SCRIPT><p>important</p>",
89                  "a lowercase <script> closed by an uppercase </SCRIPT>");
90          assertParagraphIsFound(htmlParser, "<script>var x = 1;</script ><p>important</p>",
91                  "a closing </script > with a space before the angle bracket");
92      }

94      /**
95       * FINDING 12: deeply nested HTML overflows the stack.
96       * <p>
97       *     Defect: {@code src/main/java/com/renomad/minum/htmlparsing/HtmlParseNode.java} -
98       *     {@code search} (via {@code recursiveTreeWalkSearch}), {@code print} and
99       *     {@code toString} all recurse once per level of nesting with no depth limit.  The
100      *     parser itself is iterative and happily accepts the document; only the traversal blows
101      *     up.  A document of 20,000 nested elements is about 140,000 characters - far below the
102      *     parser's own {@code MAX_HTML_SIZE} of 2,097,152 characters - and yet
103      *     {@code search()} throws {@link StackOverflowError} at the default JVM stack size.
104      *     A {@link StackOverflowError} is an {@link Error}, so it unwinds past the usual
105      *     exception handling; when the HTML being parsed came from a remote source, this is a
106      *     cheap denial of service.
107      * </p>
108      * <p>
109      *     Correct behavior: either the traversal is iterative (or depth-limited), or the parser
110      *     rejects documents nested more deeply than it can traverse.  Either way, a document
111      *     the parser accepted must be safe to search.
112      * </p>
113      * // BK: initial thought - the html parser is intended for use on trusted documents, thus invalid test.
114      */
115     @Ignore("the html parser is intended for use on trusted documents, thus invalid test.")
116     public void test_Finding_DeeplyNestedHtmlDoesNotOverflowStack() {
117         final int depth = 20_000;
118         String deeplyNestedHtml = "<b>".repeat(depth) + "hi" + "</b>".repeat(depth);
119         // well under HtmlParser.MAX_HTML_SIZE (2,097,152)
120         assertTrue(deeplyNestedHtml.length() < HtmlParser.MAX_HTML_SIZE,
121                 "this test document must be within the size the parser accepts");
122 
123         var htmlParser = new HtmlParser();
124         List<HtmlParseNode> nodes = htmlParser.parse(deeplyNestedHtml);
125         assertEquals(nodes.size(), 1);
126 
127         List<HtmlParseNode> found;
128         try {
129             found = nodes.getFirst().search(TagName.A, Map.of());
130         } catch (StackOverflowError ex) {
131             throw new AssertionError("HtmlParseNode.search should complete on a document the parser accepted.  " +
132                     "A " + deeplyNestedHtml.length() + "-character document nested " + depth + " levels deep " +
133                     "(well under MAX_HTML_SIZE of " + HtmlParser.MAX_HTML_SIZE + ") caused a StackOverflowError, " +
134                     "because recursiveTreeWalkSearch recurses once per nesting level with no depth limit.");
135         }
136 
137         assertTrue(found.isEmpty(),
138                 "there are no anchor tags in this document, so search should return an empty list");
139     }

141     private static void assertParagraphIsFound(HtmlParser htmlParser, String html, String descriptionOfHtml) {
142         List<HtmlParseNode> nodes = htmlParser.parse(html);
143         List<HtmlParseNode> paragraphs = new ArrayList<>();
144         for (HtmlParseNode node : nodes) {
145             paragraphs.addAll(node.search(TagName.P, Map.of()));
146         }
147         assertTrue(paragraphs.size() == 1,
148                 "the <p>important</p> that follows " + descriptionOfHtml + " must still be parsed, but the parser " +
149                         "returned " + nodes.size() + " top-level node(s) and " + paragraphs.size() + " paragraph(s) - " +
150                         "the closing script tag was not recognised, so the rest of the document was swallowed as " +
151                         "script text");
152         assertEquals(paragraphs.getFirst().innerText(), "important");
153     }
