70       * FINDING 1: a symbolic link inside the static-files directory escapes that directory.
71       * <p>
72       *     Defect: {@code src/main/java/com/renomad/minum/utils/FileUtils.java:120-121}
73       *     (in {@code checkFileIsWithinDirectory}).  Both the directory and the joined
74       *     full path are resolved with {@code toRealPath(LinkOption.NOFOLLOW_LINKS)}, which
75       *     means no symbolic link is ever resolved.  The containment check therefore only
76       *     compares the <em>lexical</em> path, so a symlink that lives inside the static
77       *     directory always "starts with" the static directory and passes the check - while
78       *     the subsequent read (via {@code RandomAccessFile}, which <em>does</em> follow links)
79       *     happily returns the contents of the file the link points at.  This is an arbitrary
80       *     file read out of the web root.
81       * </p>
82       * <p>
83       *     Correct behavior: {@code checkFileIsWithinDirectory} must resolve symbolic links
84       *     (i.e. use {@code toRealPath()} without {@code NOFOLLOW_LINKS}, or otherwise detect
85       *     that the target escapes) and throw {@link ForbiddenUseException} when the real
86       *     target of the requested path is outside the given directory.  This must hold for
87       *     both a symlink to a file and a symlink to a directory.
88       * </p>
89       * // BK: I have no current reason to be concerned about this.  If the developer wants to
90       * // have symbolic links in their static files directory, that is on them.
91       */

97

99              Path secretDir = Files.createDirectories(root.resolve("secret"));

102

141      * FINDING 2: an absolute STATIC_FILES_DIRECTORY makes every static file read fail.
142      * <p>
143      *     Defect: {@code src/main/java/com/renomad/minum/utils/FileReader.java:43}.
144      *     {@code readFile} calls {@code checkForBadFilePatterns} on the path it is given.
145      *     That method is a whitelist intended for the <em>untrusted fragment</em> supplied by
146      *     the remote user; among other things it rejects any path whose first character is
147      *     "/" or "\\".  But by the time {@code WebFramework.readStaticFile} calls
148      *     {@code fileReader.readFile(...)} it has already validated the fragment
149      *     ({@code WebFramework.java:632}) and joined it onto {@code STATIC_FILES_DIRECTORY}
150      *     ({@code WebFramework.java:611}).  So if an administrator configures an absolute
151      *     STATIC_FILES_DIRECTORY (e.g. {@code /var/www/static}), the joined path starts with
152      *     "/" and every single static file read throws {@link ForbiddenUseException} - which
153      *     {@code WebFramework} treats as an attack and turns into an IP ban of an innocent
154      *     visitor.
155      * </p>
156      * <p>
157      *     Correct behavior: {@code FileReader.readFile} must be able to read a legitimate,
158      *     existing file addressed by an absolute path.  The untrusted-fragment whitelist
159      *     belongs at the point where the untrusted fragment arrives, not on the fully-joined
160      *     path.
161      * </p>
162      * // BK: Removed call to checkForBadFilePatterns in FileReader#readFile.  As it turns out,
163      * // the call to this method for static files was already calling to checkForBadFilePatterns,
164      * // so the extra call was unnecessary.  Furthermore, there was a strange documentation on
165      * // two methods in IFileUtils for readBinaryFile and readTextFile - the documentation was
166      * // suggesting that it was safe to call these since they were protected by having an
167      * // internal call to checkForBadFilePatterns, but ironically also suggested that any call
168      * // of those methods needed to protect against path traversal and unallowed characters.
169      * // In other words, suggesting that it was safe to use, and also unsafe to use, at the
170      * // same time.  Removed the documentation that it was safe to use.
171      */
172     @Test
173     public void test_Finding_AbsoluteStaticDirectoryPathRejected() throws IOException {
174         Path root = Files.createTempDirectory("minum_finding_absolute_");
175         try {
176             Path staticDir = Files.createDirectories(root.resolve("static"));
177             Path indexFile = staticDir.resolve("index.html");
178             Files.writeString(indexFile, "<p>hello</p>");
179 
180             // this is what WebFramework hands to FileReader when STATIC_FILES_DIRECTORY is absolute
181             String joinedAbsolutePath = staticDir.resolve("index.html").toString();
182 
183             Map<String, byte[]> lruCache = LRUCache.getLruCache(10);
184             var fileReader = new FileReader(lruCache, false, logger);

192                         "STATIC_FILES_DIRECTORY is configured as an absolute directory).  Instead it threw " +
193                         "ForbiddenUseException: " + ex.getMessage() + " - which WebFramework escalates into an IP ban " +
194                         "of an ordinary visitor.", ex);
195             }
196 
197             assertEquals(new String(fileContents, StandardCharsets.UTF_8), "<p>hello</p>");
198         } finally {
199             deleteRecursively(root);
200         }

203     /**
204      * FINDING 3: a read/read race in the static file cache makes {@code readFile} return null.
205      * <p>
206      *     Defect: {@code src/main/java/com/renomad/minum/utils/FileReader.java:33}.
207      *     {@code readFile} tests {@code lruCache.containsKey(path)} <em>outside</em>
208      *     {@code cacheLock}, and only then takes the lock and calls {@code lruCache.get(path)}.
209      *     The cache is an {@link LRUCache} (a {@code LinkedHashMap} in access order), so any
210      *     other thread's {@code get} or {@code put} can evict the entry in between.  When that
211      *     happens {@code get} returns null and {@code readFile} returns <strong>null</strong>
212      *     to its caller (and would NPE outright if TRACE logging were enabled, since the trace
213      *     lambda dereferences {@code bytes.length}).  A null return from here becomes a
214      *     NullPointerException inside {@code WebFramework.createOkResponseForStaticFiles}, i.e.
215      *     a random 500 on a perfectly valid static asset under load.
216      * </p>
217      * <p>
218      *     Correct behavior: {@code readFile} must never return null.  The containment test and
219      *     the retrieval must happen atomically under the same lock (a single {@code get} whose
220      *     null result falls through to reading the file from disk).
221      * </p>
222      * // BK: Put the check for whether the LRUCache contains the key inside the locked area.
223      * // Also note: this was fixed, so there is no need to let this test run as many reads and
224      * // rounds as it did before, since it won't ever find anything.  I lowered the numbers so
225      * // it runs faster, but this should NEVER fail.
226      */
227     @Test
228     public void test_Finding_LruCacheReadRaceReturnsNull() throws Exception {
229         final int numberOfFiles = 40;
230         final int cacheSize = 5;
231         final int numberOfThreads = 16;
232 //        final int readsPerThread = 2_000;
233         final int readsPerThread = 500;
234         // The race is narrow, so the same experiment is repeated a few times to keep this
235         // test dependable on slower or less parallel machines.  We stop as soon as we have
236         // proof, so the usual cost of this test is a single round.
237 //        final int maximumRounds = 10;
238         final int maximumRounds = 2;
239 
240         // Note: these files must be addressed by a *relative* path, because FileReader
241         // rejects absolute paths (see test_Finding_AbsoluteStaticDirectoryPathRejected).
242         // "target" is build output and is wiped by "mvn clean".
243         Path directory = Path.of("target/findings_lru_cache_race");
244         deleteRecursively(directory);
245         Files.createDirectories(directory);
246         try {
247             List<String> paths = new ArrayList<>();
248             for (int i = 0; i < numberOfFiles; i++) {
249                 String name = "file_" + i + ".txt";
250                 Files.writeString(directory.resolve(name), "contents of file number " + i);
251                 paths.add(directory + "/" + name);
252             }
253 
254             int totalNulls = 0;
255             int totalReads = 0;
256             int roundsRun = 0;
257             for (int round = 0; round < maximumRounds && totalNulls == 0; round++) {
258                 roundsRun += 1;
259                 totalNulls += countNullsFromConcurrentReads(paths, cacheSize, numberOfThreads, readsPerThread);
260                 totalReads += numberOfThreads * readsPerThread;
261             }
262 
263             assertTrue(totalNulls == 0,
264                     "FileReader.readFile must never return null - it returned null " + totalNulls + " time(s) out of " +
265                             totalReads + " concurrent reads (" + roundsRun + " round(s)), because containsKey is checked " +
266                             "outside cacheLock and the entry can be evicted before the get inside the lock.  A null here " +
267                             "becomes a NullPointerException and a 500 response for a valid static file.");
268         } finally {
269             deleteRecursively(directory);
270         }
271     }

273     /**
274      * Runs one round of the concurrent-read experiment and returns how many times
275      * {@link FileReader#readFile(String)} handed back null.
276      */
277     private static int countNullsFromConcurrentReads(
278             List<String> paths, int cacheSize, int numberOfThreads, int readsPerThread) throws Exception {
279         Map<String, byte[]> lruCache = LRUCache.getLruCache(cacheSize);
280         // a no-op logger is used here on purpose: TestLogger serializes every log call
281         // through a single lock, which would mask the concurrency defect under test.
282         var fileReader = new FileReader(lruCache, true, new QuietLogger());
283 
284         var nullCount = new AtomicInteger(0);
285         var unexpectedException = new AtomicReference<Throwable>();
286         var startLatch = new CountDownLatch(1);
287         List<Thread> threads = new ArrayList<>();
288 
289         for (int t = 0; t < numberOfThreads; t++) {
290             final int threadIndex = t;
291             Thread thread = new Thread(() -> {
292                 try {
293                     startLatch.await();
294                     for (int i = 0; i < readsPerThread; i++) {
295                         String path = paths.get((i + threadIndex) % paths.size());
296                         byte[] result = fileReader.readFile(path);
297                         if (result == null) {
298                             nullCount.incrementAndGet();
299                         }
300                     }
301                 } catch (Throwable ex) {
302                     unexpectedException.compareAndSet(null, ex);
303                 }
304             });
305             thread.setName("lru-race-" + t);
306             threads.add(thread);
307             thread.start();
308         }
309 
310         startLatch.countDown();
311         for (Thread thread : threads) {
312             thread.join(TimeUnit.MINUTES.toMillis(2));
313         }
314 
315         Throwable thrown = unexpectedException.get();
316         assertTrue(thrown == null,
317                 "FileReader.readFile should never throw while several threads read cached static files. Instead: " + thrown);
318         return nullCount.get();
319     }

473     /**
474      * Deletes a directory tree.  Note that {@link Files#walk} does not follow
475      * symbolic links, so the links themselves get deleted rather than their targets.
476      */
477     private static void deleteRecursively(Path path) throws IOException {
478         if (!Files.exists(path, LinkOption.NOFOLLOW_LINKS)) {
479             return;
480         }
481         List<Path> allPaths;
482         try (Stream<Path> walk = Files.walk(path)) {
483             allPaths = walk.sorted(Comparator.reverseOrder()).toList();
484         }
485         for (Path p : allPaths) {
486             Files.deleteIfExists(p);
487         }
488     }
