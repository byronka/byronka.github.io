92      /**
93       * <p><b>Finding:</b> HTTP request smuggling through {@code Transfer-Encoding: chunked}.</p>
94       * <p><b>Defect:</b> {@code WebFramework.java:385} - {@code determineIfKeepAlive} only guards
95       * against an unread body when a {@code content-length} header was present
96       * ({@code request.getHeaders().contentLength() >= 0}).  A request that declares
97       * {@code Transfer-Encoding: chunked} has no content-length, so the guard never fires; the
98       * body is left unread in the socket and keep-alive stays on.  The framework then reads the
99       * unconsumed body as if it were the next request on the connection.</p>
100      * <p><b>Correct behavior:</b> Minum does not implement chunked request decoding, so a request
101      * with {@code Transfer-Encoding: chunked} must either be rejected, or at minimum must close
102      * the connection afterwards, so that the unread body can never be interpreted as a second,
103      * smuggled request.</p>
104      * // BK: This is an invalid test, the AI does not seem to understand the big picture.
105      * // If a client sends us a request without a content length, like is being done here,
106      * // yes it is true we will move on to the next bytes as simply a new request.
107      *
108      * // But, there is no special security concern with that.
109      *
110      * // If any particular request provides invalid input, we deal with it.
111      * // The system does not handle chunked encoded requests,
112      * // and does not act as a proxy server, so the issues being raised here have no impact.
113      *
114      * // In other words, I could put this as a question: What actual security concern will
115      * // exist if users are able to send consecutive requests to the Minum server?
116      * // Answer: No more concern than normal.
117      */
118     @Ignore("Not an actual security concern")
119     public void test_Finding_ChunkedRequestSmuggling() throws Exception {
120         var wf = new WebFramework(context, default_zdt);
121         var webEngine = new WebEngine(context, wf);
122 
123         // an endpoint that does NOT read the body
124         wf.registerPath(POST, "noop", request -> Response.htmlOk("NOOP_OK"));
125         // an endpoint that must NOT be reachable by the smuggled request
126         wf.registerPath(GET, "smuggled", request -> Response.htmlOk("SMUGGLED_MARKER_XYZ"));
127 
128         String allBytes;
129         try (IServer primaryServer = webEngine.startServer()) {

404     /**

413      * partition header (a non-ASCII filename, for instance) makes {@code countBytesRead} drift
414      * below the true number of bytes consumed.</p>
415      * <p><b>Correct behavior:</b> byte accounting must use the UTF-8 byte length of each line, so
416      * that a body with a correct, byte-accurate Content-Length parses successfully no matter what
417      * characters appear in the partition headers.</p>

419      *        stringValue.getBytes(StandardCharsets.UTF_8).length

424         String multiByteFilename = "漢".repeat(100);
425         String body =
426                 "--i_am_a_boundary\r\n" +
427                 "Content-Disposition: form-data; name=\"myfile\"; filename=\"" + multiByteFilename + ".txt\"\r\n" +
428                 "Content-Type: text/plain\r\n" +
429                 "\r\n" +
430                 "abcdef\r\n" +
431                 "--i_am_a_boundary--\r\n";
432         byte[] bodyBytes = body.getBytes(StandardCharsets.UTF_8);
433 
434         var bodyProcessor = new BodyProcessor(context);
435 
436         // Content-Length is byte-accurate, exactly as a real client would send it
437         Body bodyResult = bodyProcessor.extractBodyFromInputStream(

441 
442         assertEquals(bodyResult.getBodyType(), BodyType.MULTIPART);
443         assertEqualByteArray(

478                 "Content-Disposition: form-data; name=\"text1\"\r\n" +
479                 "\r\n" +
480                 "abcdef\r\n" +
481                 "--abc--\r\n";
482         byte[] bodyBytes = body.getBytes(StandardCharsets.UTF_8);
483         socketWrapper.is = new ByteArrayInputStream(bodyBytes);
484 
485         IRequest request = new Request(
486                 new Headers(List.of(
487                         "content-length: " + bodyBytes.length,
488                         "content-type: multipart/form-data; boundary=abc; charset=utf-8")),
489                 RequestLine.EMPTY,
490                 "456.456.456.456",
491                 socketWrapper,
492                 new BodyProcessor(context),
493                 false);
494 
495         StreamingMultipartPartition partition = request.getMultipartIterable().iterator().next();
496 
497         assertEquals(partition.getContentDisposition().getName(), "text1");
498         assertTrue("abcdef".equals(new String(partition.readAllBytes(), StandardCharsets.UTF_8)),

509     /**
510      * <p><b>Finding:</b> {@code Content-Length: +5} is accepted as the value 5.</p>

512      * {@code Long.parseLong}, which accepts an optional leading {@code '+'} (and would also
513      * accept a leading {@code '-'}, though the subsequent negative check catches that one).</p>

515      * {@code 1*DIGIT} - digits only.  A value of {@code +5} is malformed and must be rejected
516      * with a {@link BadRequestException}.  Accepting it invites request-smuggling
517      * disagreements with any intermediary that parses the header strictly.</p>
518      * // BK: Added a check by a new helper method, StringUtils.isPureDigits, to check
519      * that the input consists solely of digits.  It is still necessary to check the
520      * conversion to a long though.
521      */
522     @Test
523     public void test_Finding_ContentLengthRejectsPlusPrefix() {
524         Headers headers = new Headers(List.of("Content-Length: +5"));
525         assertThrows(BadRequestException.class, headers::contentLength);
526     }

531      * ==========================================================================

533 
534     /**
535      * <p><b>Finding:</b> {@link FullSystem#shutdown()} never stops {@code TheBrig}, so the
536      * brig's database directory stays registered in the {@link Context} forever.</p>
537      * <p><b>Defect:</b> {@code FullSystem.java:227-248} - {@code shutdown()} calls
538      * {@code closeCore}, which closes the servers, kills the action queues and deletes the
539      * SYSTEM_RUNNING file, but never calls {@code theBrig.stop()}.  {@code TheBrig.stop()}
540      * (TheBrig.java:121-127) is what calls {@code inmatesDb.stop()}, and
541      * {@code DbEngine2.stop()} (DbEngine2.java:519-522) is what calls
542      * {@code context.removeFromPaths(dbDirectory)} and flushes pending writes to disk.</p>
543      * <p><b>Correct behavior:</b> after a full shutdown, the brig's database must have been
544      * stopped: its data flushed and its path de-registered, so that a subsequent
545      * {@link FullSystem} using the same context/directory can start without tripping the
546      * "Attempted to register more than one database to the same path" guard in
547      * {@code AbstractDb} (AbstractDb.java:100).</p>
548      * // BK: Added code in FullSystem#closeCore to explicitly stop the brig
549      */
550     @Test
551     public void test_Finding_TheBrigIsStoppedOnShutdown() throws IOException {
552         Path scratchDb = Path.of("/tmp/minum_findings_test_db_" + System.nanoTime());
553         Properties properties = Constants.getConfiguredProperties();
554         properties.setProperty("DB_DIRECTORY", scratchDb.toString());
555         properties.setProperty("IS_THE_BRIG_ENABLED", "true");
556         properties.setProperty("ENABLE_SYSTEM_RUNNING_MARKER", "false");
557         properties.setProperty("SERVER_PORT", "8085");
558         properties.setProperty("SSL_SERVER_PORT", "8446");
559 
560         Context myContext = buildTestingContext("FindingsWebTests - brig shutdown", properties);
561         try {
562             var fullSystem = new FullSystem(myContext);
563             fullSystem.start();
564             Path brigPath = Path.of(myContext.getConstants().dbDirectory, "the_brig");
565             assertTrue(myContext.isDbPathRegistered(brigPath),
566                     "Sanity check: while running, the brig's database path should be registered");
567 
568             fullSystem.shutdown();
569             MyThread.sleep(50);
570 
571             assertFalse(myContext.isDbPathRegistered(brigPath),
572                     "FullSystem.shutdown() must stop TheBrig, which stops its database, which de-registers its " +
573                             "path (" + brigPath + ").  Because theBrig.stop() is never called, the path stays " +
574                             "registered and the brig's data is never flushed - a second FullSystem on the same " +
575                             "context/directory would fail with 'Attempted to register more than one database to " +
576                             "the same path'.");
577         } finally {
578             TestFramework.shutdownTestingContext(myContext);
579             MyThread.sleep(SERVER_CLOSE_WAIT_TIME);
580         }
581     }

589     /**
590      * Read everything the server sends until the socket closes or goes quiet.
591      * The socket must have had a soTimeout set.
592      */
593     private static String readUntilTimeoutOrClose(InputStream is) throws IOException {
594         var baos = new ByteArrayOutputStream();
595         byte[] buffer = new byte[1024];
596         try {
597             while (true) {
598                 int numRead = is.read(buffer);
599                 if (numRead == -1) break;
600                 baos.write(buffer, 0, numRead);
601             }
602         } catch (SocketTimeoutException ignored) {
603             // this is how we know the server has said all it intends to say
604         }
605         return baos.toString(StandardCharsets.UTF_8);
606     }
